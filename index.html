/** * DP Pass Checkout — Google Apps Script backend * * Stores every checkout in a Google Sheet, serves pass availability to the * web page, and emails people when their pass is due soon or overdue. * * SETUP (one time): * 1. Open the Google Sheet you want to use → Extensions → Apps Script. * Replace everything in Code.gs with this file and Save. * 2. Edit the CONFIG block below (admin email, admin key, timing). * 3. Run `setup` once from the editor (choose it in the function dropdown, * click Run, and approve the permissions). This creates the sheet * headers and the hourly email-reminder trigger. * 4. Deploy → Manage deployments → edit your existing Web app deployment → * Version: "New version" → Deploy. (Execute as: Me, Who has access: * Anyone.) Editing the existing deployment keeps the same /exec URL. */ // ===================== CONFIG ===================== var CONFIG = { SHEET_NAME: 'Checkouts', // Optional: the ID of the Google Sheet to use (the long code in its URL between /d/ and /edit). // Leave '' and the script uses the Sheet it's attached to, or creates "DP Pass Checkout Log" // in this Google account's Drive the first time it runs. SHEET_ID: '', PASSES: ['349', '350', '351', '352', '353'], MAX_DURATION_MS: 7 * 24 * 60 * 60 * 1000, // 7 days // Password the admin types into the Admin dashboard to confirm returns / renew. // CHANGE THIS before deploying. ADMIN_KEY: '05121984', // Who gets new-request notices and a copy of overdue notices. Leave '' to skip. ADMIN_EMAIL: 'staciaf@uga.edu', // When anyone hits Reply on one of these emails, it goes here instead of the Gmail account. REPLY_TO: 'staciaf@uga.edu', // Send the "due soon" reminder this long before the expected return time. DUE_SOON_MS: 24 * 60 * 60 * 1000, // 24 hours // After a pass is overdue, repeat the overdue email this often. OVERDUE_REPEAT_MS: 24 * 60 * 60 * 1000, // daily // Email the requester a confirmation when they reserve a pass. SEND_CONFIRMATION: true, // Where people pick up and return the pass (used in emails). PICKUP_LOCATION: "the Dean's office", // Link back to the checkout page, included in emails. PAGE_URL: 'https://zappa103.github.io/DP-Pass-Checkout/', TIME_ZONE: 'America/New_York' }; var HEADERS = [ 'id', 'pass', 'name', 'email', 'reason', 'out', 'expectedBack', 'status', 'requestedAt', 'actualReturnAt', 'renewals', 'dueSoonSentAt', 'overdueSentAt' ]; // ================================================== // ---------- one-time setup ---------- function setup() { getSheet_(); Logger.log('Data is stored in: ' + getSpreadsheet_().getUrl()); ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'sendReminders') ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger('sendReminders').timeBased().everyHours(1).create(); Logger.log('Sheet ready and hourly reminder trigger installed.'); } // ---------- web endpoints ---------- /** * GET → public availability only. No names, emails, reasons or pass numbers, * just which time windows are already booked. */ function doGet() { var booked = readAll_().filter(function (r) { return r.status === 'reserved'; }).map(function (r) { return { slot: CONFIG.PASSES.indexOf(r.pass), out: r.out, expectedBack: r.expectedBack }; }); return json_({ passes: CONFIG.PASSES.length, booked: booked }); } /** POST → { action: 'create' | 'checkKey' | 'list' | 'return' | 'renew', ... } */ function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); var action = body.action || 'create'; if (action === 'create') return json_(create_(body)); if (action === 'checkKey') return json_({ ok: body.adminKey === CONFIG.ADMIN_KEY }); // Everything below is admin-only. if (body.adminKey !== CONFIG.ADMIN_KEY) return json_({ ok: false, error: 'Admin key is wrong.' }); if (action === 'list') return json_({ ok: true, rows: readAll_() }); if (action === 'return') return json_(markReturned_(body.id)); if (action === 'renew') return json_(renew_(body.id)); return json_({ ok: false, error: 'Unknown action.' }); } catch (err) { return json_({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } // ---------- actions ---------- function create_(b) { var name = String(b.name || '').trim(); var email = String(b.email || '').trim(); var reason = String(b.reason || '').trim(); var out = new Date(b.out); var back = new Date(b.expectedBack); if (!name || !reason || !email) return { ok: false, error: 'Name, email and reason are required.' }; if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return { ok: false, error: 'That email address doesn\'t look right.' }; if (isNaN(out) || isNaN(back)) return { ok: false, error: 'Please choose a pick-up and return time.' }; if (back <= out) return { ok: false, error: 'The return time has to be after the pick-up time.' }; if (back - out > CONFIG.MAX_DURATION_MS) return { ok: false, error: 'Passes can be reserved for up to 7 days.' }; var rows = readAll_(); var pass = findFreePass_(rows, out.getTime(), back.getTime(), null); if (!pass) return { ok: false, error: 'All passes are booked for those times. Please try different times.' }; var now = new Date(); var id = Utilities.getUuid(); var rec = { id: id, pass: pass, name: name, email: email, reason: reason, out: out, expectedBack: back, status: 'reserved', requestedAt: now, actualReturnAt: '', renewals: '[]', dueSoonSentAt: '', overdueSentAt: '' }; getSheet_().appendRow(HEADERS.map(function (h) { return rec[h]; })); if (CONFIG.SEND_CONFIRMATION) { sendMail_(email, 'Your departmental parking pass is reserved', 'Hi ' + name + ',\n\n' + 'Your departmental parking pass is reserved.\n\n' + ' Pass: DP-' + pass + '\n' + ' Your scheduled reservation time: ' + fmt_(out) + '\n' + ' Please return it by: ' + fmt_(back) + '\n\n' + 'Pick up the pass at ' + CONFIG.PICKUP_LOCATION + ' at your scheduled reservation time, ' + 'and return it there when you\'re done.\n\n' + 'Please follow UGA Transportation & Parking Services\' rules for Departmental Permits.\n'); } if (CONFIG.ADMIN_EMAIL) { sendMail_(CONFIG.ADMIN_EMAIL, 'New pass request: DP-' + pass + ' for ' + name, name + ' (' + email + ') reserved DP-' + pass + '.\n\n' + ' Reservation time: ' + fmt_(out) + '\n' + ' Return by: ' + fmt_(back) + '\n' + ' Reason: ' + reason + '\n\n' + CONFIG.PAGE_URL); } return { ok: true }; } function markReturned_(id) { var found = findRow_(id); if (!found) return { ok: false, error: 'Checkout not found.' }; setCell_(found.rowIndex, 'status', 'returned'); setCell_(found.rowIndex, 'actualReturnAt', new Date()); return { ok: true }; } function renew_(id) { var found = findRow_(id); if (!found) return { ok: false, error: 'Checkout not found.' }; var r = found.record; if (r.status !== 'reserved') return { ok: false, error: 'Only active checkouts can be renewed.' }; var oldBack = new Date(r.expectedBack); var newBack = new Date(oldBack.getTime() + CONFIG.MAX_DURATION_MS); var others = readAll_().filter(function (o) { return o.id !== id; }); var conflict = others.some(function (o) { return o.status === 'reserved' && o.pass === r.pass && overlaps_(oldBack.getTime(), newBack.getTime(), new Date(o.out).getTime(), effectiveEnd_(o)); }); if (conflict) return { ok: false, error: 'DP-' + r.pass + ' is already booked by someone else in the next 7 days.' }; var renewals = parseJson_(r.renewals); renewals.push({ at: new Date().toISOString(), previousTimeBack: oldBack.toISOString(), newTimeBack: newBack.toISOString() }); setCell_(found.rowIndex, 'expectedBack', newBack); setCell_(found.rowIndex, 'renewals', JSON.stringify(renewals)); setCell_(found.rowIndex, 'dueSoonSentAt', ''); // so they get a fresh reminder setCell_(found.rowIndex, 'overdueSentAt', ''); if (r.email) { sendMail_(r.email, 'Your parking pass DP-' + r.pass + ' has been renewed', 'Hi ' + r.name + ',\n\n' + 'Your parking pass DP-' + r.pass + ' has been renewed.\n' + 'Please return it to ' + CONFIG.PICKUP_LOCATION + ' by ' + fmt_(newBack) + '.\n'); } return { ok: true, expectedBack: newBack.toISOString() }; } // ---------- reminder emails (runs hourly via trigger) ---------- function sendReminders() { var now = Date.now(); readAll_().forEach(function (r, i) { if (r.status !== 'reserved' || !r.email) return; var out = new Date(r.out).getTime(); var back = new Date(r.expectedBack).getTime(); if (isNaN(back) || now < out) return; // reservation hasn't started yet var rowIndex = i + 2; if (now > back) { var last = r.overdueSentAt ? new Date(r.overdueSentAt).getTime() : 0; if (!last || now - last >= CONFIG.OVERDUE_REPEAT_MS) { sendMail_(r.email, 'Overdue: please return parking pass DP-' + r.pass, 'Hi ' + r.name + ',\n\n' + 'Parking pass DP-' + r.pass + ' was due back ' + fmt_(new Date(back)) + ' and hasn\'t been returned yet.\n\n' + 'Please return it to ' + CONFIG.PICKUP_LOCATION + ' as soon as possible so the next person can use it. ' + 'If you still need it, reply to this email to ask about a renewal.\n', CONFIG.ADMIN_EMAIL); setCell_(rowIndex, 'overdueSentAt', new Date()); } } else if (back - now <= CONFIG.DUE_SOON_MS && !r.dueSoonSentAt) { sendMail_(r.email, 'Reminder: parking pass DP-' + r.pass + ' is due ' + fmt_(new Date(back)), 'Hi ' + r.name + ',\n\n' + 'Just a reminder that parking pass DP-' + r.pass + ' is due back at ' + CONFIG.PICKUP_LOCATION + ' by ' + fmt_(new Date(back)) + '.\n\n' + 'Need it longer? Reply to this email before it\'s due to ask about a renewal.\n'); setCell_(rowIndex, 'dueSoonSentAt', new Date()); } }); } // ---------- helpers ---------- function getSpreadsheet_() { if (CONFIG.SHEET_ID) return SpreadsheetApp.openById(CONFIG.SHEET_ID); var active = SpreadsheetApp.getActiveSpreadsheet(); if (active) return active; var props = PropertiesService.getScriptProperties(); var id = props.getProperty('SHEET_ID'); if (id) return SpreadsheetApp.openById(id); var created = SpreadsheetApp.create('DP Pass Checkout Log'); props.setProperty('SHEET_ID', created.getId()); return created; } function getSheet_() { var ss = getSpreadsheet_(); var sh = ss.getSheetByName(CONFIG.SHEET_NAME) || ss.insertSheet(CONFIG.SHEET_NAME); if (sh.getLastRow() === 0) { sh.appendRow(HEADERS); sh.setFrozenRows(1); } return sh; } function readAll_() { var sh = getSheet_(); var values = sh.getDataRange().getValues(); if (values.length < 2) return []; var head = values[0].map(String); return values.slice(1).filter(function (row) { return row.join('') !== ''; }).map(function (row) { var o = {}; head.forEach(function (h, i) { var v = row[i]; o[h] = v instanceof Date ? v.toISOString() : (v === null || v === undefined ? '' : String(v)); }); o.renewals = o.renewals || '[]'; return o; }); } function findRow_(id) { var rows = readAll_(); for (var i = 0; i < rows.length; i++) { if (rows[i].id === String(id)) return { rowIndex: i + 2, record: rows[i] }; } return null; } function setCell_(rowIndex, header, value) { var col = HEADERS.indexOf(header) + 1; getSheet_().getRange(rowIndex, col).setValue(value); } // A pass that's past due but not returned is treated as out for at least the next hour. function effectiveEnd_(r) { var end = new Date(r.expectedBack).getTime(); return r.status === 'reserved' ? Math.max(end, Date.now() + 60 * 60 * 1000) : end; } function findFreePass_(rows, start, end, ignoreId) { var active = rows.filter(function (r) { return r.status === 'reserved' && r.id !== ignoreId; }); for (var i = 0; i < CONFIG.PASSES.length; i++) { var p = CONFIG.PASSES[i]; var taken = active.some(function (r) { return r.pass === p && overlaps_(start, end, new Date(r.out).getTime(), effectiveEnd_(r)); }); if (!taken) return p; } return null; } function overlaps_(aS, aE, bS, bE) { return aS < bE && bS < aE; } function parseJson_(s) { try { return JSON.parse(s) || []; } catch (e) { return []; } } function fmt_(d) { return Utilities.formatDate(d, CONFIG.TIME_ZONE, 'EEE MMM d, h:mm a'); } function sendMail_(to, subject, body, cc) { try { var opts = { name: 'DP Pass Checkout' }; if (CONFIG.REPLY_TO) opts.replyTo = CONFIG.REPLY_TO; if (cc) opts.cc = cc; MailApp.sendEmail(to, subject, body, opts); } catch (err) { console.error('Email failed to ' + to + ': ' + err); } } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
